{{- /* ========================================================================== JSON-LD (GEO / AEO / AI-SEO) Emits a single @graph per page so answer engines and LLM crawlers can resolve the publisher entity, the site, the page, the breadcrumb trail, and — where a post has one — its FAQ block: Organization (every page) -> stable publisher identity, optional sameAs WebSite (every page) -> site entity, language, publisher link BreadcrumbList(non-home) -> Home > Section > Page BlogPosting (posts) -> headline, dates, author, keywords, section FAQPage (posts w/ FAQ)-> auto-derived from the "Frequently Asked Questions" headings in the markdown source Replaces Hugo's _internal/schema.html, whose output was orphaned (no itemscope/itemtype wrapper) and therefore invalid. =========================================================================== */ -}} {{- $orgURL := site.Params.organizationURL | default "https://nvnest.com" -}} {{- $orgID := printf "%s#organization" $orgURL -}} {{- $siteID := printf "%s#website" site.BaseURL -}} {{- /* Publisher entity */ -}} {{- $org := dict "@type" "Organization" "@id" $orgID "name" site.Title "url" $orgURL -}} {{- with site.Params.slogan -}} {{- $org = merge $org (dict "slogan" .) -}} {{- end -}} {{- with site.Params.organizationDescription -}} {{- $org = merge $org (dict "description" .) -}} {{- end -}} {{- with site.Params.organizationLogo -}} {{- $org = merge $org (dict "logo" (absURL .)) -}} {{- end -}} {{- with site.Params.sameAs -}} {{- $org = merge $org (dict "sameAs" .) -}} {{- end -}} {{- /* Site entity */ -}} {{- $website := dict "@type" "WebSite" "@id" $siteID "url" site.BaseURL "name" site.Title "inLanguage" site.LanguageCode "publisher" (dict "@id" $orgID) -}} {{- $graph := slice $org $website -}} {{- /* Breadcrumb trail (skip on the home page — a single-item trail is noise) */ -}} {{- if not .IsHome -}} {{- $crumbs := slice (dict "@type" "ListItem" "position" 1 "name" "Home" "item" site.BaseURL) -}} {{- $section := .CurrentSection -}} {{- if and $section (ne $section.Permalink .Permalink) (ne $section.Permalink site.BaseURL) -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" (add (len $crumbs) 1) "name" $section.Title "item" $section.Permalink) -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" (add (len $crumbs) 1) "name" .Title "item" .Permalink) -}} {{- $graph = $graph | append (dict "@type" "BreadcrumbList" "@id" (printf "%s#breadcrumb" .Permalink) "itemListElement" $crumbs) -}} {{- end -}} {{- /* Article node for content pages (BlogPosting for posts, plain WebPage for standalone pages such as /about/) */ -}} {{- if and (eq .Kind "page") (in site.Params.mainSections .Section) -}} {{- $author := dict "@id" $orgID -}} {{- with .Params.author -}} {{- $author = dict "@type" "Person" "name" . -}} {{- end -}} {{- $post := dict "@type" "BlogPosting" "@id" (printf "%s#article" .Permalink) "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "headline" .Title "description" (.Description | default (.Summary | plainify)) "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "inLanguage" site.LanguageCode "author" $author "publisher" (dict "@id" $orgID) "isPartOf" (dict "@id" $siteID) "wordCount" .WordCount "url" .Permalink -}} {{- with .Params.tags -}} {{- $post = merge $post (dict "keywords" (delimit . ", ")) -}} {{- end -}} {{- with .Params.categories -}} {{- $post = merge $post (dict "articleSection" (delimit . ", ")) -}} {{- end -}} {{- $graph = $graph | append $post -}} {{- else if eq .Kind "page" -}} {{- $graph = $graph | append (dict "@type" "WebPage" "@id" (printf "%s#webpage" .Permalink) "url" .Permalink "name" .Title "description" (.Description | default (.Summary | plainify)) "isPartOf" (dict "@id" $siteID) "inLanguage" site.LanguageCode "publisher" (dict "@id" $orgID) ) -}} {{- end -}} {{- /* FAQ node, derived from the post's "Frequently Asked Questions" section */ -}} {{- $faqPairs := partial "faq-pairs.html" . -}} {{- if $faqPairs -}} {{- $questions := slice -}} {{- range $faqPairs -}} {{- $questions = $questions | append (dict "@type" "Question" "name" .q "acceptedAnswer" (dict "@type" "Answer" "text" .a) ) -}} {{- end -}} {{- $graph = $graph | append (dict "@type" "FAQPage" "@id" (printf "%s#faq" $.Permalink) "mainEntity" $questions ) -}} {{- end -}}
